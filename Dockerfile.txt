FROM php:8.2-apache

# Install system dependencies and MySQL extensions
RUN apt-get update && apt-get install -y \
    libzip-dev \
    zip \
    && docker-php-ext-install pdo pdo_mysql mysqli

# Enable Apache mod_rewrite
RUN a2enmod rewrite

# Copy project files into Apache web directory
COPY . /var/www/html/

# Set permissions for uploads and admin reviews
RUN chown -R www-data:www-data /var/www/html \
    && chmod -R 755 /var/www/html

# Render dynamically assigns a port, configure Apache to listen to it
ENV PORT=10000
RUN sed -i 's/80/${PORT}/g' /etc/apache2/ports.conf /etc/apache2/sites-available/000-default.conf

EXPOSE 10000
